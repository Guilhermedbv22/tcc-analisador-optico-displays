/*
==========================================================
 TCC - Medidor de Características de Displays
 Versão 1.0 - Aquisição de Dados

 Função:
 - Timer de Hardware
 - Interrupção a cada 100 us
 - Leitura do ADC
 - Buffer de Amostras
 - Envio Serial

 Desenvolvido para ESP32
==========================================================
*/

// ----------------------------
// Bibliotecas
// ----------------------------
#include <Arduino.h>

// ----------------------------
// Configurações
// ----------------------------

// GPIO onde está ligado o amplificador
const int PINO_SENSOR = 34;

// Intervalo entre amostras (100 us)
const uint32_t TEMPO_AMOSTRAGEM_US = 100;

// Quantidade de amostras por aquisição
const uint16_t TAM_BUFFER = 1000;

// ----------------------------
// Variáveis Globais
// ----------------------------

// Buffer para armazenar as leituras
volatile uint16_t buffer[TAM_BUFFER];

// Índice atual do buffer
volatile uint16_t indice = 0;

// Informa quando o buffer está cheio
volatile bool bufferCheio = false;

// Ponteiro para o Timer
hw_timer_t *timer = NULL;

// ----------------------------
// Interrupção do Timer
// ----------------------------
void IRAM_ATTR onTimer()
{
    // Se ainda existe espaço no buffer
    if (indice < TAM_BUFFER)
    {
        // Lê o ADC
        buffer[indice] = analogRead(PINO_SENSOR);

        // Próxima posição
        indice++;
    }

    // Buffer completo
    else
    {
        bufferCheio = true;

        // Reinicia para a próxima aquisição
        indice = 0;
    }
}

// ----------------------------
// Setup
// ----------------------------
void setup()
{
    Serial.begin(115200);

    delay(1000);

    Serial.println();
    Serial.println("==================================");
    Serial.println(" Sistema de Aquisição Inicializado");
    Serial.println("==================================");

    // Cria Timer de Hardware
    timer = timerBegin(1000000);

    // Liga a interrupção
    timerAttachInterrupt(timer, &onTimer);

    // Dispara a cada 100 us
    timerAlarm(timer, TEMPO_AMOSTRAGEM_US, true, 0);
}

// ----------------------------
// Loop Principal
// ----------------------------
void loop()
{
    // Espera o buffer encher
    if (bufferCheio)
    {
        bufferCheio = false;

        Serial.println("----- NOVA AQUISICAO -----");

        for (int i = 0; i < TAM_BUFFER; i++)
        {
            Serial.println(buffer[i]);
        }

        Serial.println("----- FIM -----");
    }
}
